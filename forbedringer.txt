import React, { useState, useMemo } from 'react';
import { 
  Briefcase, 
  Calendar, 
  Clock, 
  AlertTriangle, 
  CheckCircle2, 
  Info, 
  UserCheck, 
  GraduationCap,
  Building2,
  ArrowRight,
  ShieldAlert,
  ExternalLink
} from 'lucide-react';


const App = () => {
  // --- STATE ---
  const [totalMonths, setTotalMonths] = useState(42); 
  const [employmentType, setEmploymentType] = useState('employee'); 
  const [isCurrentlyEmployed, setIsCurrentlyEmployed] = useState(true);
  const [weeklyHoursNow, setWeeklyHoursNow] = useState(37);


  // --- LOGIK ---
  const analysis = useMemo(() => {
    let messages = [];
    let isEligible = true;
    let warnings = [];
    
    // Beregn historik-progress (max 100%)
    let historyProgress = Math.min(100, (totalMonths / 42) * 100);


    // 1. Undtagelser (Pensionist)
    if (employmentType === 'pensioner') {
      return {
        isEligible: true,
        status: 'Klar (Undtaget)',
        description: 'Som folkepensionist eller førtidspensionist opfylder du automatisk kravet om beskæftigelse.',
        color: 'text-emerald-600 bg-emerald-50 border-emerald-200',
        progress: 100,
        messages: [],
        warnings: []
      };
    }


    // 2. Historik-tjek (3 år og 6 måneder = 42 måneder)
    if (totalMonths < 42) {
      isEligible = false;
      messages.push(`Du mangler ${42 - totalMonths} måneders fuldtidsarbejde (min. 120t/md) inden for de sidste 4 år.`);
    }


    // 3. Nu-kravet (Aktuel beskæftigelse og timetal)
    let currentJobOk = true;
    if (!isCurrentlyEmployed) {
      isEligible = false;
      currentJobOk = false;
      messages.push('Du skal være i beskæftigelse på det tidspunkt, hvor lovforslaget fremsættes.');
    } else if (weeklyHoursNow < 15) {
      isEligible = false;
      currentJobOk = false;
      messages.push('Dit nuværende arbejde skal være på mindst 15 timer om ugen.');
    }


    // Justering af Progress Bar
    let finalProgress = historyProgress;
    if (historyProgress === 100 && !currentJobOk) {
      finalProgress = 95; 
    }


    if (employmentType === 'self-employed') {
      warnings.push("Som selvstændig gælder der særlige dokumentationskrav for overskud og omsætning.");
    }
    if (employmentType === 'student') {
      warnings.push("Kun erhvervsuddannelser (elevplads) tæller som beskæftigelse.");
    }


    return {
      isEligible,
      status: isEligible ? 'Du opfylder kravene' : 'Du mangler lidt endnu',
      messages,
      warnings,
      progress: finalProgress,
      color: isEligible ? 'text-emerald-700 bg-emerald-50 border-emerald-200' : 'text-amber-700 bg-amber-50 border-amber-200'
    };
  }, [totalMonths, employmentType, isCurrentlyEmployed, weeklyHoursNow]);


  return (
    <div className="min-h-screen bg-slate-50 text-slate-900 font-sans p-4 md:p-8">
      <div className="max-w-5xl mx-auto">
        
        <header className="mb-8 border-b border-slate-200 pb-6 text-center md:text-left">
          <div className="flex items-center justify-center md:justify-start gap-2 text-blue-600 font-bold text-xs uppercase tracking-widest mb-2">
            <Briefcase size={16} />
            <span>Indfødsret-Guide</span>
          </div>
          <h1 className="text-3xl font-black text-slate-900 tracking-tight text-balance">
            Tjek Beskæftigelse
          </h1>
          <p className="text-slate-500 mt-1">Tjek dine arbejdskrav for dansk statsborgerskab.</p>
        </header>


        <div className="grid lg:grid-cols-12 gap-8">
          
          {/* INPUT KOLONNE */}
          <div className="lg:col-span-7 space-y-6">
            
            <section className="bg-white rounded-3xl p-6 shadow-sm border border-slate-200">
              <h2 className="font-bold mb-6 flex items-center gap-2 text-slate-800">
                <Clock size={18} className="text-blue-500" />
                Arbejdshistorik (Sidste 4 år)
              </h2>


              <div className="space-y-8">
                <div>
                  <div className="flex justify-between items-end mb-4">
                    <label className="text-sm font-bold text-slate-700 leading-tight">
                      Måneder med fuldtidsarbejde <br/>
                      <span className="text-[10px] text-slate-400 font-normal uppercase tracking-wide">(Mindst 120 timer pr. måned)</span>
                    </label>
                    <span className="text-2xl font-black text-blue-600 leading-none">{totalMonths} <span className="text-xs text-slate-400">mdr.</span></span>
                  </div>
                  <input 
                    type="range" min="0" max="48" value={totalMonths}
                    onChange={(e) => setTotalMonths(parseInt(e.target.value))}
                    className="w-full h-2 bg-slate-100 rounded-lg appearance-none cursor-pointer accent-blue-600"
                  />
                </div>


                <div className="grid grid-cols-2 gap-3">
                  {[
                    { id: 'employee', label: 'Lønmodtager', icon: <UserCheck size={14}/> },
                    { id: 'self-employed', label: 'Selvstændig', icon: <Building2 size={14}/> },
                    { id: 'student', label: 'Elev / Lærling', icon: <GraduationCap size={14}/> },
                    { id: 'pensioner', label: 'Pensionist', icon: <Info size={14}/> }
                  ].map(type => (
                    <button
                      key={type.id}
                      onClick={() => setEmploymentType(type.id)}
                      className={`flex items-center gap-2 p-4 rounded-2xl text-xs font-bold border-2 transition-all ${
                        employmentType === type.id ? 'border-blue-600 bg-blue-50 text-blue-700 shadow-sm' : 'border-slate-100 bg-white text-slate-500 hover:border-slate-200'
                      }`}
                    >
                      {type.icon}
                      {type.label}
                    </button>
                  ))}
                </div>
              </div>
            </section>


            <section className="bg-white rounded-3xl p-6 shadow-sm border border-slate-200">
              <h2 className="font-bold mb-6 flex items-center gap-2 text-slate-800">
                <Calendar size={18} className="text-blue-500" />
                Nuværende Status
              </h2>
              <div className="space-y-6">
                <div className="flex items-center justify-between p-4 bg-slate-50 rounded-2xl">
                  <span className="text-sm font-bold text-slate-700">Er du i arbejde lige nu?</span>
                  <button 
                    onClick={() => setIsCurrentlyEmployed(!isCurrentlyEmployed)}
                    className={`w-14 h-7 rounded-full transition-colors relative ${isCurrentlyEmployed ? 'bg-blue-600' : 'bg-slate-300'}`}
                  >
                    <div className={`absolute top-1 w-5 h-5 bg-white rounded-full shadow-sm transition-all ${isCurrentlyEmployed ? 'left-8' : 'left-1'}`} />
                  </button>
                </div>


                {isCurrentlyEmployed && (
                  <div className="animate-in fade-in slide-in-from-top-2">
                    <div className="flex justify-between text-sm font-bold mb-3">
                      <span className="text-slate-700">Ugentlige timer i nuværende job</span>
                      <span className="text-blue-600">{weeklyHoursNow} timer</span>
                    </div>
                    <input 
                      type="range" min="0" max="50" value={weeklyHoursNow}
                      onChange={(e) => setWeeklyHoursNow(parseInt(e.target.value))}
                      className="w-full h-2 bg-slate-100 rounded-lg appearance-none cursor-pointer accent-blue-600"
                    />
                  </div>
                )}
              </div>
            </section>


            {/* ANSVARSFRASKRIVELSE SEKTION */}
            <footer className="mt-8 p-6 bg-slate-100 rounded-3xl border border-slate-200 space-y-3">
              <div className="flex items-center gap-2 text-slate-600">
                <ShieldAlert size={18} className="shrink-0 text-slate-400" />
                <h4 className="text-xs font-black uppercase tracking-wider">Ansvarsfraskrivelse</h4>
              </div>
              <p className="text-[11px] leading-relaxed text-slate-500 font-medium">
                Dette værktøj er udelukkende beregnet som en vejledende simulator baseret på de nuværende regler om indfødsret. 
                Beregningerne udgør ikke juridisk rådgivning eller en garanti for statsborgerskab. 
                Ministeriet foretager altid en individuel vurdering af din sag, herunder dokumentation for løntimer, feriepenge og ansættelsesforhold.
              </p>
              <div className="flex flex-wrap gap-4 pt-2">
                <a 
                  href="https://uim.dk/statsborgerskab" 
                  target="_blank" 
                  rel="noopener noreferrer"
                  className="text-[10px] flex items-center gap-1 font-bold text-blue-600 hover:underline"
                >
                  Besøg UIM.dk <ExternalLink size={10} />
                </a>
              </div>
            </footer>
          </div>


          {/* RESULTAT KOLONNE */}
          <div className="lg:col-span-5">
            <div className={`rounded-[2.5rem] p-8 border-2 transition-all sticky top-8 ${analysis.color}`}>
              <div className="flex justify-between items-start mb-6">
                <div>
                  <p className="text-[10px] font-black uppercase tracking-widest opacity-60 mb-1">Vurdering</p>
                  <h3 className="text-2xl font-black leading-tight">{analysis.status}</h3>
                </div>
                <div className="p-3 bg-white/50 rounded-2xl shadow-sm">
                  {analysis.isEligible ? <CheckCircle2 size={32} className="text-emerald-600" /> : <AlertTriangle size={32} className="text-amber-600" />}
                </div>
              </div>


              <div className="space-y-6">
                <div className="space-y-2">
                  <div className="flex justify-between text-[10px] font-black uppercase">
                    <span>Samlet status</span>
                    <span>{analysis.progress.toFixed(0)}%</span>
                  </div>
                  <div className="h-4 w-full bg-white/40 rounded-full overflow-hidden p-1 shadow-inner text-emerald-500">
                    <div className="h-full bg-current rounded-full transition-all duration-1000" style={{ width: `${analysis.progress}%` }} />
                  </div>
                </div>


                <div className="space-y-4">
                  {analysis.description && (
                    <p className="text-sm font-bold leading-relaxed">{analysis.description}</p>
                  )}
                  
                  {analysis.messages.length > 0 ? (
                    analysis.messages.map((msg, i) => (
                      <div key={i} className="flex gap-3 text-sm font-bold leading-tight">
                        <ArrowRight size={16} className="mt-0.5 shrink-0" />
                        {msg}
                      </div>
                    ))
                  ) : (
                    !analysis.description && (
                      <div className="flex gap-3 text-sm font-bold leading-tight">
                        <CheckCircle2 size={16} className="mt-0.5 shrink-0" />
                        Baseret på dine svar opfylder du kravene til beskæftigelse.
                      </div>
                    )
                  )}


                  {analysis.warnings.length > 0 && (
                    <div className="mt-4 pt-4 border-t border-current/10 space-y-2">
                      <p className="text-[10px] font-black uppercase opacity-60">Vigtigt at vide:</p>
                      {analysis.warnings.map((warn, i) => (
                        <div key={i} className="flex gap-2 text-xs font-semibold italic opacity-80">
                          <Info size={14} className="shrink-0" />
                          {warn}
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              </div>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};


export default App;



KARENS BEREGNER



import React, { useState, useMemo } from 'react';
import { 
  Calendar, 
  Info, 
  AlertTriangle, 
  CheckCircle2, 
  Clock, 
  Calculator, 
  Shield, 
  Scale, 
  History,
  TrendingUp,
  FileText,
  AlertCircle
} from 'lucide-react';


const App = () => {
  // --- STATE ---
  const [penaltyType, setPenaltyType] = useState('bode_over_3000');
  const [penaltyDate, setPenaltyDate] = useState(new Date().toISOString().split('T')[0]);
  const [isRecidivism, setIsRecidivism] = useState(false);
  const [hasCalculated, setHasCalculated] = useState(false);


  // --- BEREGNINGSLOGIK (Baseret på din specifikke kode) ---
  const calculation = useMemo(() => {
    if (!penaltyDate && penaltyType !== 'sigtet') return null;


    const startDate = new Date(penaltyDate);
    let resultText = "";
    let explanation = "";
    let status = "OK";
    let isPermanent = false;
    let isBlocked = false;
    let karensYears = 0;
    let recidivismYears = isRecidivism ? 3 : 0;


    switch(penaltyType) {
        case 'bode_under_3000':
            if (isRecidivism) {
                karensYears = 3;
                explanation = "Bøder under 3.000 kr. giver normalt ikke karens, men ved gentagelsestilfælde kan der pålægges en karensperiode (her estimeret til 3 år).";
                status = "WARNING";
            } else {
                resultText = "Ingen karens";
                explanation = "Bøder under 3.000 kr. (undtagen visse særlove) medfører normalt ikke karensperiode.";
                status = "OK";
            }
            break;
        case 'bode_over_3000':
            karensYears = 4.5 + recidivismYears;
            explanation = `Bøder på 3.000 kr. eller derover giver 4,5 års karens. ${isRecidivism ? 'Da det er et gentagelsestilfælde, er der lagt 3 år til.' : ''}`;
            status = "WARNING";
            break;
        case 'fart_over_3000':
            karensYears = 4.5 + recidivismYears;
            explanation = `Fartbøder følger standarden for bøder. ${isRecidivism ? 'Gentagelse af trafikforseelser vægter tungt og lægger 3 år til perioden.' : ''}`;
            status = "WARNING";
            break;
        case 'betinget':
            karensYears = 12 + recidivismYears;
            explanation = `Betinget fængsel giver 12 års karens. ${isRecidivism ? 'Som gentagelsestilfælde forlænges din karens med yderligere 3 år.' : ''}`;
            status = "DANGER";
            break;
        case 'ubetinget':
            isPermanent = true;
            resultText = "Permanent udelukket";
            explanation = "Ubetinget fængsel medfører som udgangspunkt livsvarig udelukkelse fra dansk statsborgerskab jf. § 19, stk. 4.";
            status = "CRITICAL";
            break;
        case 'sigtet':
            isBlocked = true;
            resultText = "Du kan ikke søge";
            explanation = "Du kan ikke optages på et lovforslag om naturalisation, så længe du er under sigtelse.";
            status = "CRITICAL";
            break;
        default: break;
    }


    let finalDate = null;
    let progressPercent = 0;
    let remainingDays = 0;


    if (!isPermanent && !isBlocked) {
        if (status === "OK" && !isRecidivism) {
            progressPercent = 100;
        } else {
            finalDate = new Date(startDate);
            const totalMonths = karensYears * 12;
            finalDate.setMonth(finalDate.getMonth() + totalMonths);
            
            const now = new Date();
            const totalDays = (finalDate - startDate) / (1000 * 60 * 60 * 24);
            const passedDays = (now - startDate) / (1000 * 60 * 60 * 24);
            
            progressPercent = Math.min(100, Math.max(0, Math.round((passedDays / totalDays) * 100)));
            remainingDays = Math.max(0, Math.round(totalDays - passedDays));
            
            const options = { year: 'numeric', month: 'long', day: 'numeric' };
            resultText = "Kan søge: " + finalDate.toLocaleDateString('da-DK', options);
        }
    }


    return {
      resultText,
      explanation,
      status,
      isPermanent,
      isBlocked,
      progressPercent,
      remainingDays,
      finalDateString: finalDate?.toLocaleDateString('da-DK', { day: 'numeric', month: 'long', year: 'numeric' })
    };
  }, [penaltyType, penaltyDate, isRecidivism]);


  const getStatusColor = (status) => {
    switch(status) {
      case 'OK': return 'text-emerald-600 bg-emerald-50 border-emerald-100';
      case 'WARNING': return 'text-amber-600 bg-amber-50 border-amber-100';
      case 'DANGER': return 'text-orange-600 bg-orange-50 border-orange-100';
      case 'CRITICAL': return 'text-red-600 bg-red-50 border-red-100';
      default: return 'text-slate-600 bg-slate-50 border-slate-100';
    }
  };


  return (
    <div className="min-h-screen bg-[#F8FAFC] text-slate-900 font-sans p-4 md:p-8">
      <div className="max-w-5xl mx-auto">
        
        {/* Header */}
        <header className="text-center mb-12">
            <div className="inline-flex items-center gap-2 bg-blue-50 text-blue-700 px-4 py-1.5 rounded-full text-xs font-bold uppercase tracking-widest mb-4">
                <Scale size={14} />
                <span>Officielle Cirkulæreskrivelse Regler</span>
            </div>
            <h1 className="text-4xl font-black text-slate-900 tracking-tight">Vandel & Karens-Beregner</h1>
            <p className="text-slate-500 mt-2 max-w-xl mx-auto">Tjek om din historik påvirker din mulighed for dansk statsborgerskab.</p>
        </header>


        <div className="grid lg:grid-cols-12 gap-8 items-start">
          
          {/* INPUT SEKTION */}
          <div className="lg:col-span-5 space-y-6">
            <section className="bg-white rounded-[2rem] p-8 shadow-xl shadow-slate-200/40 border border-slate-100">
              <div className="flex items-center gap-3 mb-8">
                <div className="p-3 bg-blue-600 rounded-2xl text-white shadow-lg shadow-blue-200">
                  <Calculator size={20} />
                </div>
                <h2 className="text-xl font-bold">Indtast oplysninger</h2>
              </div>


              <div className="space-y-6">
                {/* Type */}
                <div className="space-y-2">
                  <label className="text-xs font-black text-slate-400 uppercase ml-1">Straf/Afgørelse</label>
                  <select 
                    value={penaltyType}
                    onChange={(e) => {setPenaltyType(e.target.value); setHasCalculated(false);}}
                    className="w-full bg-slate-50 border-2 border-slate-100 rounded-2xl px-4 py-4 focus:border-blue-500 transition-all outline-none font-semibold text-slate-700"
                  >
                    <option value="bode_under_3000">Bøde under 3.000 kr.</option>
                    <option value="bode_over_3000">Bøde på 3.000 kr. eller derover</option>
                    <option value="fart_over_3000">Fartbøde (3.000 kr.+)</option>
                    <option value="betinget">Betinget fængsel</option>
                    <option value="ubetinget">Ubetinget fængsel</option>
                    <option value="sigtet">Jeg er pt. under sigtelse</option>
                  </select>
                </div>


                {/* Dato */}
                {penaltyType !== 'sigtet' && (
                  <div className="space-y-2 animate-in fade-in slide-in-from-top-2">
                    <label className="text-xs font-black text-slate-400 uppercase ml-1">Domsdato / Bødedato</label>
                    <input 
                      type="date" 
                      value={penaltyDate}
                      onChange={(e) => {setPenaltyDate(e.target.value); setHasCalculated(false);}}
                      className="w-full bg-slate-50 border-2 border-slate-100 rounded-2xl px-4 py-4 focus:border-blue-500 transition-all outline-none font-semibold"
                    />
                  </div>
                )}


                {/* Gentagelse */}
                <div 
                  onClick={() => {setIsRecidivism(!isRecidivism); setHasCalculated(false);}}
                  className={`p-5 rounded-2xl border-2 cursor-pointer transition-all flex items-center gap-4 ${
                    isRecidivism ? 'border-blue-600 bg-blue-50/50' : 'border-slate-100 bg-white hover:border-slate-200'
                  }`}
                >
                  <div className={`w-6 h-6 rounded-md border-2 flex items-center justify-center transition-colors ${
                    isRecidivism ? 'bg-blue-600 border-blue-600' : 'border-slate-200 bg-white'
                  }`}>
                    {isRecidivism && <CheckCircle2 size={16} className="text-white" />}
                  </div>
                  <div>
                    <p className="text-sm font-bold text-slate-800">Gentagelsestilfælde?</p>
                    <p className="text-[10px] text-slate-500 font-bold uppercase tracking-tighter">Lægger typisk 3 år til karens</p>
                  </div>
                </div>


                <button 
                  onClick={() => setHasCalculated(true)}
                  className="w-full bg-slate-900 text-white font-black py-5 rounded-2xl hover:bg-blue-600 transition-all shadow-xl shadow-slate-200 active:scale-[0.98] uppercase tracking-widest text-sm"
                >
                  Beregn Karensperiode
                </button>
              </div>
            </section>


            <div className="bg-slate-900 rounded-[2rem] p-8 text-white">
              <div className="flex items-center gap-3 mb-4">
                <AlertCircle size={20} className="text-blue-400" />
                <h3 className="font-bold text-sm uppercase">Vigtig viden</h3>
              </div>
              <p className="text-slate-400 text-xs leading-relaxed font-medium">
                Reglerne om vandel (god opførsel) er strenge. Selv mindre forseelser kan have opsættende virkning, hvis de sker gentagne gange eller under en eksisterende karensperiode.
              </p>
            </div>
          </div>


          {/* RESULTAT SEKTION */}
          <div className="lg:col-span-7">
            {!hasCalculated ? (
              <div className="h-full min-h-[400px] border-4 border-dashed border-slate-200 rounded-[3rem] flex flex-col items-center justify-center p-12 text-center opacity-60">
                <div className="w-20 h-20 bg-slate-100 rounded-full flex items-center justify-center mb-6 text-slate-300">
                  <History size={40} />
                </div>
                <h3 className="text-xl font-bold text-slate-400 italic">Venter på beregning...</h3>
                <p className="text-slate-400 text-sm mt-2">Udfyld formularen til venstre for at se din status.</p>
              </div>
            ) : (
              <div className="space-y-6 animate-in fade-in zoom-in-95 duration-500">
                {/* Result Card */}
                <div className="bg-white rounded-[2.5rem] p-10 shadow-2xl shadow-slate-200 border border-slate-100 relative overflow-hidden">
                  <div className="relative z-10">
                    <div className="flex justify-between items-start mb-8">
                      <div>
                        <p className="text-[10px] font-black text-slate-400 uppercase tracking-[0.2em] mb-2">Ansøgnings-status</p>
                        <h2 className="text-3xl font-black text-slate-900 leading-tight">
                          {calculation.resultText}
                        </h2>
                      </div>
                      <div className={`p-4 rounded-2xl ${getStatusColor(calculation.status)}`}>
                        {calculation.status === 'OK' ? <CheckCircle2 size={32} /> : <Clock size={32} />}
                      </div>
                    </div>


                    <div className="grid grid-cols-2 gap-4 mb-8">
                       <div className="bg-slate-50 p-5 rounded-2xl border border-slate-100">
                          <p className="text-[10px] font-bold text-slate-400 uppercase mb-1">Mål-dato</p>
                          <p className="font-black text-slate-800 text-lg">
                            {calculation.isPermanent || calculation.isBlocked ? 'Ingen dato' : (calculation.finalDateString || 'Allerede klar')}
                          </p>
                       </div>
                       <div className="bg-slate-50 p-5 rounded-2xl border border-slate-100">
                          <p className="text-[10px] font-bold text-slate-400 uppercase mb-1">Status</p>
                          <div className={`text-xs font-black uppercase px-2 py-1 rounded inline-block ${getStatusColor(calculation.status)}`}>
                            {calculation.status}
                          </div>
                       </div>
                    </div>


                    {/* Progress Bar - "Gennemført tid" */}
                    {!calculation.isPermanent && !calculation.isBlocked && (
                      <div className="space-y-4">
                        <div className="flex justify-between items-end">
                          <div className="flex items-center gap-2">
                            <TrendingUp size={16} className="text-blue-500" />
                            <span className="text-xs font-black text-slate-600 uppercase">Gennemført karens-tid</span>
                          </div>
                          <span className="text-2xl font-black text-slate-900">{calculation.progressPercent}%</span>
                        </div>
                        <div className="h-4 w-full bg-slate-100 rounded-full overflow-hidden p-1 shadow-inner">
                          <div 
                            className={`h-full rounded-full transition-all duration-1000 ease-out ${
                              calculation.progressPercent === 100 ? 'bg-emerald-500' : 'bg-blue-600'
                            }`}
                            style={{ width: `${calculation.progressPercent}%` }}
                          />
                        </div>
                        {calculation.progressPercent < 100 && (
                          <p className="text-center text-[10px] font-bold text-slate-400 uppercase tracking-widest pt-2">
                            Ca. {calculation.remainingDays} dage resterende
                          </p>
                        )}
                      </div>
                    )}
                  </div>
                </div>


                {/* Explanation Box */}
                <div className="bg-white rounded-3xl p-8 border border-slate-100 shadow-sm flex items-start gap-5">
                  <div className="p-3 bg-blue-50 rounded-xl text-blue-600 shrink-0">
                    <FileText size={24} />
                  </div>
                  <div>
                    <h4 className="font-bold text-slate-800 mb-2">Juridisk Grundlag</h4>
                    <p className="text-sm text-slate-500 leading-relaxed italic">
                      "{calculation.explanation}"
                    </p>
                  </div>
                </div>


                {/* Info Note */}
                <div className="bg-amber-50 rounded-3xl p-6 border border-amber-100 flex items-start gap-4">
                  <AlertTriangle size={20} className="text-amber-600 shrink-0 mt-0.5" />
                  <p className="text-xs text-amber-800 font-medium leading-relaxed">
                    <strong>Bemærk:</strong> Beregningen følger de generelle retningslinjer i cirkulæreskrivelsen. Der tages ikke højde for personlige forhold, benådninger eller særregler for visse typer kriminalitet.
                  </p>
                </div>
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  );
};


export default App;

            document.getElementById('paywall').style.display = 'none';

        function initYearSelector() {
            const container = document.getElementById('yearSelector');
            for (let i = 0; i < 12; i++) {
                const year = new Date().getFullYear() - i;
                const btn = document.createElement('button');
                btn.className = `year-btn ${year === currentYear ? 'active' : ''}`;
                btn.innerText = year;
                btn.onclick = () => setYear(year, btn);
                container.appendChild(btn);
            }
        }


        function setYear(year, btn) {
            currentYear = year;
            document.querySelectorAll('.year-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            const d = document.getElementById('departureDate');
            const r = document.getElementById('returnDate');
            const suffix = (val) => val && val.includes('-') ? val.substring(4) : "-07-01";
            d.value = year + suffix(d.value);
            r.value = year + suffix(r.value);
        }


        function addTrip() {
            const dVal = document.getElementById('departureDate').value;
            const rVal = document.getElementById('returnDate').value;
            if (!dVal || !rVal) return;
            const d = new Date(dVal);
            const r = new Date(rVal);
            if (r < d) return alert("Hjemkomst skal være efter afrejse");
            
            const days = Math.ceil(Math.abs(r - d) / (1000 * 60 * 60 * 24));
            trips.push({ id: Date.now(), start: dVal, end: rVal, days: days, year: d.getFullYear() });
            trips.sort((a, b) => new Date(b.start) - new Date(a.start));
            
            document.getElementById('departureDate').value = '';
            document.getElementById('returnDate').value = '';
            updateUI();
        }


        function removeTrip(id) {
            trips = trips.filter(t => t.id !== id);
            updateUI();
        }


        function updateUI() {
            const list = document.getElementById('tripList');
            const printList = document.getElementById('printListContent');
            const printRules = document.getElementById('printRuleStatus');
            list.innerHTML = ''; printList.innerHTML = ''; printRules.innerHTML = '';
            
            let total = 0;
            let yearStats = {};
            let violations = { longTrip: false, yearLimit: false, frequency: false };


            trips.forEach(t => {
                total += t.days;
                if (!yearStats[t.year]) yearStats[t.year] = { days: 0, count: 0 };
                yearStats[t.year].days += t.days;
                yearStats[t.year].count++;


                if (t.days > 28) violations.longTrip = true;
                if (yearStats[t.year].days > 42) violations.yearLimit = true;
                if (yearStats[t.year].count >= 10) violations.frequency = true;


                const div = document.createElement('div');
                div.className = `flex justify-between items-center p-4 bg-slate-900 border ${t.days > 28 ? 'border-red-500/50' : 'border-slate-700'} rounded-xl`;
                div.innerHTML = `
                    <div>
                        <p class="text-white font-bold text-sm">${t.start} — ${t.end}</p>

                    </div>
                    <button onclick="removeTrip(${t.id})" class="text-slate-600 hover:text-red-400 text-[10px] font-black tracking-widest uppercase transition-colors">Slet</button>
                `;
                list.appendChild(div);


                printList.insertAdjacentHTML('beforeend', `<div class="flex justify-between border-b py-2"><span>${t.start} til ${t.end}</span><span class="font-bold">${t.days} dage</span></div>`);
            });


            // Update Rule Cards
            updateRule('rule4weeks', 'rule4text', violations.longTrip, "Over 28 dage!", "Rejser under 28 dage");
            updateRule('rule6weeks', 'rule6text', violations.yearLimit, "Over 42 dage årligt!", "Under 6 uger årligt");
            updateRule('ruleFrequency', 'ruleFreqtext', violations.frequency, "For mange rejser (10+)", "Normal frekvens");


            // Print rule summary
            printRules.innerHTML += `<li>4-ugers reglen: ${violations.longTrip ? 'BRUDT' : 'OK'}</li>`;
            printRules.innerHTML += `<li>6-ugers grænsen: ${violations.yearLimit ? 'OVERSKREDET' : 'OK'}</li>`;
            printRules.innerHTML += `<li>Rejsefrekvens: ${violations.frequency ? 'ADVARSEL' : 'OK'}</li>`;


            document.getElementById('totalDays').innerText = total + " dage";
            document.getElementById('printTotal').innerText = total + " dage";
            
            const statusEl = document.getElementById('globalStatus');
            if (trips.length === 0) {
                statusEl.innerText = "Ingen data";
                statusEl.className = "inline-block px-3 py-1 bg-slate-800 rounded-full text-xs font-bold text-slate-400 border border-slate-700";
            } else if (violations.longTrip || violations.yearLimit) {
                statusEl.innerText = "Risiko for afslag";
                statusEl.className = "inline-block px-3 py-1 bg-red-500 text-white rounded-full text-xs font-bold border border-red-400 shadow-lg shadow-red-900/20";
            } else {
                statusEl.innerText = "Sikkert ophold";
                statusEl.className = "inline-block px-3 py-1 bg-emerald-600 text-white rounded-full text-xs font-bold border border-emerald-400 shadow-lg shadow-emerald-900/20";
            }
            
            if(trips.length === 0) list.innerHTML = '<p class="text-slate-600 text-sm italic text-center py-10 tracking-wide">Indtast rejser for at starte analysen...</p>';
        }


        function updateRule(cardId, textId, isViolated, badMsg, goodMsg) {
            const card = document.getElementById(cardId);
            const text = document.getElementById(textId);
            if (trips.length === 0) {
                card.className = "rule-card bg-slate-900/40 p-4 rounded-xl border border-slate-700";

                text.className = "text-[11px] mt-1 text-red-400 font-bold";
            } else {
                card.className = "rule-card rule-ok p-4 rounded-xl border-emerald-500/30";
                text.innerText = "✓ " + goodMsg;
                text.className = "text-[11px] mt-1 text-emerald-500 font-medium";
